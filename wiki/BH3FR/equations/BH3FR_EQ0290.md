# Equation BH3FR_EQ0290

$$
\begin{aligned} & \boldsymbol{S}\left(\operatorname{GRAD}_{L} \phi \partial \bar{N}\right) ; n=\text { const, } \quad \underset{\Omega(L)}{\boldsymbol{S}} \operatorname{DIV}_{L} \bar{\phi} \partial V= \\ & =\underset{\Omega(L-1)}{\boldsymbol{S}} \bar{\phi} \partial \bar{V}, \quad \text { SSROT }{ }_{L} \bar{\phi} \partial^{2} \bar{F}=\boldsymbol{S} \varnothing ð \bar{N} \\ & \partial \bar{N}=\sum_{i=1}^{L} \partial_{i} \bar{Z}(i), \quad \partial V=\prod_{k=1}^{L} \partial_{k} Z(k) \\ & \partial \bar{V}=\sum_{j=1}^{L} \bar{e}_{j} \partial V_{j}, \quad \partial V_{j}=\prod_{k=1}^{j-1} \partial_{k} Z(k) \prod_{j+1}^{L} \partial_{k} Z(k), \\ & \left(\bar{e}_{i} \bar{e}_{k}\right)_{L}=\hat{E}, \quad \partial^{2} \bar{F}=\left[\partial_{i} Z(i) \partial_{k} Z(k)\right]_{L} \end{aligned}
$$
