# Equation WDorg4_EQ0038

$$
\begin{aligned} & \mathrm{w}_{0}=\mathrm{a}_{1} \\ & \mathrm{w}_{1}=\left(\mathrm{o}_{1}\right)\left(\mathrm{b}_{1}, \mathrm{~b}_{2}, \mathrm{~b}_{3}^{*}\right)=\mathrm{b}_{1} \mathrm{o}_{1} \mathrm{~b}_{2} \mathrm{o}_{1} \mathrm{~b}_{3}^{*} \\ & \mathrm{w}_{2}=\left(\mathrm{o}_{2} \mathrm{o}_{1}\right)\left(\left(\mathrm{c}_{1}\right),\left(\mathrm{c}_{1}, \mathrm{c}_{2}\right),\left(\mathrm{c}_{1}, \mathrm{c}_{2}, \mathrm{c}_{3}\right)\right)=\mathrm{c}_{1} \mathrm{o}_{2}\left(\mathrm{c}_{1} \mathrm{o}_{1} \mathrm{o}_{2}\right) \mathrm{o}_{2}\left(\mathrm{c}_{1} \mathrm{o}_{1} \mathrm{c}_{2} \mathrm{o}_{1} \mathrm{c}_{3}\right) \\ & \mathrm{w}_{3}=\left(\mathrm{o}_{3} \mathrm{o}_{2} \mathrm{o}_{1}\right)\left(\left(\left(\mathrm{d}_{1}\right)\right),\left(\left(\mathrm{d}_{1}\right),\left(\mathrm{d}_{1}, \mathrm{~d}_{2}\right)\right)\right)=\mathrm{d}_{1} \mathrm{o}_{3}\left(\mathrm{~d}_{1} \mathrm{o}_{2}\left(\mathrm{~d}_{1} \mathrm{o}_{1} \mathrm{~d}_{2}\right)\right) \\ & \mathrm{w}_{4}=\left(\mathrm{o}_{4} \mathrm{o}_{3} \mathrm{o}_{2} \mathrm{o}_{1}\right)\left(\left(\left(\left(\mathrm{e}_{1}\right)\right)\right),\left(\left(\left(\mathrm{e}_{1}\right)\right),\left(\left(\mathrm{e}_{1}\right),\left(\mathrm{e}_{1}, \mathrm{e}_{2}\right)\right)\right), \ldots .\right) \end{aligned}
$$
