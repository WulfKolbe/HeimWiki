# Equation WDorg4_EQ0045

$$
\begin{aligned} & \left(a_{1}\right) \subset\left(a_{1}, a_{2}\right) \subset\left(a_{1}, a_{2}, a_{3}\right) \subset \ldots \\ & \left(\left(a_{1}\right)\right) \subset\left(\left(a_{1}\right),\left(a_{1}, a_{2}\right)\right) \subset\left(\left(a_{1}\right),\left(a_{1}, a_{2}\right),\left(a_{1}, a_{2}, a_{3}\right)\right) \subset \ldots \\ & \left(\left(\left(a_{1}\right)\right)\right) \subset\left(\left(\left(a_{1}\right)\right),\left(\left(a_{1}\right),\left(a_{1}, a_{2}\right)\right)\right) \subset \ldots \\ & \left.\left(\left(\left(\left(a_{1}\right)\right)\right)\right) \subset\left(\left(\left(\left(a_{1}\right)\right)\right),\left(\left(a_{1}\right)\right),\left(\left(a_{1}\right),\left(a_{1}, a_{2}\right)\right)\right)\right) \subset \ldots \\ & \left.\left(\left(\left(\left(\left(a_{1}\right)\right)\right)\right)\right) \subset\left(\left(\left(\left(a_{1}\right)\right)\right)\right),\left(\left(\left(\left(a_{1}\right)\right)\right),\left(\left(\left(a_{1}\right)\right),\left(\left(a_{1}\right),\left(a_{1}, a_{2}\right)\right)\right)\right)\right) \subset \ldots \end{aligned}
